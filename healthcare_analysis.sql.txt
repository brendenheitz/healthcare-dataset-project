SELECT COUNT(*) AS total_rows FROM healthcare_raw;

-- Find duplicate admissions
SELECT Name, Age, "Date of Admission", Hospital, COUNT(*) AS times_appearing
FROM healthcare_raw
GROUP BY Name, Age, "Date of Admission", Hospital
HAVING COUNT(*) > 1;
-- Result: 534 duplicate groups


-- Check Gender consistency
SELECT Gender, COUNT(*) AS count
FROM healthcare_raw
GROUP BY Gender;
-- Result: clean (Male / Female only)


-- Check for bad billing
SELECT COUNT(*) AS bad_billing
FROM healthcare_raw
WHERE "Billing Amount" < 0 OR "Billing Amount" IS NULL;
-- Result: 108 bad records


-- Check for impossible dates
SELECT COUNT(*) AS bad_dates
FROM healthcare_raw
WHERE "Discharge Date" < "Date of Admission";
-- Result: 0


-- Date range
SELECT MIN("Date of Admission") AS earliest, MAX("Date of Admission") AS latest

-- Result: 2019-05-08 to 2024-05-07




-- -----------------------------------------------------------------------------
-- STEP 2: DATA CLEANING & TRANSFORMATION
-- -----------------------------------------------------------------------------


DROP TABLE IF EXISTS healthcare_clean;


CREATE TABLE healthcare_clean AS
SELECT
    TRIM(Name) AS name,
    Age AS age,
    TRIM(Gender) AS gender,
    UPPER(TRIM("Blood Type")) AS blood_type,
    TRIM("Medical Condition") AS medical_condition,
    "Date of Admission" AS date_of_admission,
    TRIM(Doctor) AS doctor,
    TRIM(Hospital) AS hospital,
    TRIM("Insurance Provider") AS insurance_provider,
    ROUND("Billing Amount", 2) AS billing_amount,
    "Room Number" AS room_number,
    TRIM("Admission Type") AS admission_type,
    "Discharge Date" AS discharge_date,
    TRIM(Medication) AS medication,
    TRIM("Test Results") AS test_results,
    -- DERIVED COLUMNS
    CAST(julianday("Discharge Date") - julianday("Date of Admission") AS INT) AS length_of_stay,
    CASE
        WHEN Age < 18 THEN '0-17'
        WHEN Age BETWEEN 18 AND 30 THEN '18-30'
        WHEN Age BETWEEN 31 AND 45 THEN '31-45'
        WHEN Age BETWEEN 46 AND 60 THEN '46-60'
        WHEN Age BETWEEN 61 AND 75 THEN '61-75'
        ELSE '76+'
    END AS age_band,
    CAST(strftime('%Y', "Date of Admission") AS INT) AS admission_year,
    CAST(strftime('%m', "Date of Admission") AS INT) AS admission_month
FROM healthcare_raw
WHERE Name IS NOT NULL
  AND TRIM(Name) <> ''
  AND Age BETWEEN 0 AND 120
  AND "Billing Amount" >= 0
  AND "Billing Amount" IS NOT NULL
  AND "Discharge Date" >= "Date of Admission";


-- Remove duplicates
DELETE FROM healthcare_clean
WHERE rowid NOT IN (
    SELECT MIN(rowid)
    FROM healthcare_clean
    GROUP BY name, age, gender, date_of_admission, hospital
);


-- Verify
SELECT COUNT(*) FROM healthcare_clean;
-- Result: 54,860 clean rows (640 removed: 108 bad billing + 532 duplicates)




-------------------------------------------------------------------------------
-- STEP 3: ANALYTICAL QUERIES (feeds the Power BI dashboard)
-------------------------------------------------------------------------------


-- Headline KPIs
SELECT
    COUNT(*) AS total_patients,
    COUNT(DISTINCT hospital) AS total_hospitals,
    COUNT(DISTINCT doctor) AS total_doctors,
    ROUND(SUM(billing_amount), 2) AS total_revenue,
    ROUND(AVG(billing_amount), 2) AS avg_billing,
    ROUND(AVG(length_of_stay), 2) AS avg_length_of_stay,
    ROUND(AVG(age), 1) AS avg_age
FROM healthcare_clean;
-- Total revenue: $1.40 billion | Avg LOS: 15.5 days | Avg age: 51.5


-- Patients by medical condition
SELECT
    medical_condition,
    COUNT(*) AS patient_count,
    ROUND(AVG(billing_amount), 2) AS avg_billing,
    ROUND(SUM(billing_amount), 2) AS total_billing,
    ROUND(AVG(length_of_stay), 2) AS avg_los,
    SUM(CASE WHEN test_results = 'Abnormal' THEN 1 ELSE 0 END) AS abnormal_count,
    ROUND(100.0 * SUM(CASE WHEN test_results = 'Abnormal' THEN 1 ELSE 0 END) / COUNT(*), 2) AS abnormal_pct
FROM healthcare_clean
GROUP BY medical_condition
ORDER BY patient_count DESC;


-- Monthly admissions trend
SELECT
    admission_year,
    admission_month,
    COUNT(*) AS admissions,
    ROUND(SUM(billing_amount), 2) AS revenue
FROM healthcare_clean
GROUP BY admission_year, admission_month
ORDER BY admission_year, admission_month;


-- Demographics
SELECT
    age_band,
    gender,
    COUNT(*) AS patient_count,
    ROUND(AVG(billing_amount), 2) AS avg_billing,
    ROUND(AVG(length_of_stay), 2) AS avg_los
FROM healthcare_clean
GROUP BY age_band, gender
ORDER BY age_band, gender;


-- Admission type distribution
SELECT
    admission_type,
    COUNT(*) AS patient_count,
    ROUND(AVG(billing_amount), 2) AS avg_billing,
    ROUND(AVG(length_of_stay), 2) AS avg_los
FROM healthcare_clean
GROUP BY admission_type
ORDER BY patient_count DESC;


-- Insurance provider revenue
SELECT
    insurance_provider,
    COUNT(*) AS patient_count,
    ROUND(SUM(billing_amount), 2) AS total_revenue,
    ROUND(AVG(billing_amount), 2) AS avg_claim
FROM healthcare_clean
GROUP BY insurance_provider
ORDER BY total_revenue DESC;
-- Cigna leads at $284.3M


-- Top 10 hospitals by revenue
SELECT
    hospital,
    COUNT(*) AS patient_count,
    ROUND(SUM(billing_amount), 2) AS total_revenue,
    ROUND(AVG(length_of_stay), 2) AS avg_los
FROM healthcare_clean
GROUP BY hospital
ORDER BY total_revenue DESC
LIMIT 10;


-- Test results by condition
SELECT
    medical_condition,
    test_results,
    COUNT(*) AS patient_count
FROM healthcare_clean
GROUP BY medical_condition, test_results
ORDER BY medical_condition, test_results;
